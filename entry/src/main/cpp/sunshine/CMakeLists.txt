cmake_minimum_required(VERSION 3.16)

set(EXPERIMENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../experiments/sunshine")
set(EXPERIMENT_DEPS "$ENV{STAR_SUNSHINE_DEPS}")
if(NOT EXISTS "${EXPERIMENT_DEPS}/moonlight-common-c/CMakeLists.txt")
    message(FATAL_ERROR "Prepare the pinned Sunshine experiment dependencies first (see experiments/sunshine/README.md).")
endif()
set(CORE_DIR "${EXPERIMENT_DEPS}/moonlight-common-c")
set(OPENSSL_ROOT_DIR "${EXPERIMENT_DEPS}/ohos-openssl/prelude/${CMAKE_OHOS_ARCH_ABI}")
set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT_DIR}/include")
set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libcrypto.a")
set(OPENSSL_SSL_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libssl.a")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
if(CMAKE_OHOS_ARCH_ABI STREQUAL "x86_64")
    set(SUNSHINE_PEER_ARCH x86_64-unknown-linux-ohos)
else()
    set(SUNSHINE_PEER_ARCH aarch64-unknown-linux-ohos)
endif()
set(SUNSHINE_PEER_ARCHIVE "${EXPERIMENT_ROOT}/peer/target/${SUNSHINE_PEER_ARCH}/release/libsesame_wan_peer.a")
if(NOT EXISTS "${SUNSHINE_PEER_ARCHIVE}")
    message(FATAL_ERROR "Build the locked Sunshine UDP peer for ${SUNSHINE_PEER_ARCH} first.")
endif()
add_library(sunshine_wan_peer STATIC IMPORTED)
set_target_properties(sunshine_wan_peer PROPERTIES IMPORTED_LOCATION "${SUNSHINE_PEER_ARCHIVE}")
add_subdirectory("${CORE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/moonlight-core")
target_compile_options(moonlight-common-c PRIVATE -Wno-unused-command-line-argument)

add_library(sunshine_stream_exp SHARED bridge.cpp client.cpp)
if("$ENV{SESAME_SUNSHINE_LAN_BENCH}" STREQUAL "1")
    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_LAN_BENCH=1)
endif()
if("$ENV{SESAME_SUNSHINE_LONG_BENCH}" STREQUAL "1")
    if(NOT "$ENV{SESAME_SUNSHINE_LAN_BENCH}" STREQUAL "1")
        message(FATAL_ERROR "The long session lease is limited to the isolated LAN-bench debug build.")
    endif()
    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_LONG_BENCH=1)
endif()
if("$ENV{SESAME_SUNSHINE_COMPANION}" STREQUAL "1")
    if(NOT "$ENV{SESAME_SUNSHINE_LAN_BENCH}" STREQUAL "1")
        message(FATAL_ERROR "The companion prototype is restricted to the isolated LAN debug build.")
    endif()
    target_sources(sunshine_stream_exp PRIVATE companion.cpp stream_audio.cpp)
    target_link_libraries(sunshine_stream_exp PRIVATE ohaudio Opus::opus)
    target_include_directories(sunshine_stream_exp PRIVATE "${OPENSSL_INCLUDE_DIR}")
    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_COMPANION=1)
    set(COMPANION_BROKER "$ENV{SESAME_SUNSHINE_COMPANION_BROKER}")
    if("$ENV{SESAME_SUNSHINE_USB_LAN_CANDIDATE}" STREQUAL "1")
        if(NOT "$ENV{SESAME_STREAM_TEST_BUILD}" STREQUAL "1" OR NOT "$ENV{SESAME_SUNSHINE_LONG_BENCH}" STREQUAL "1" OR NOT COMPANION_BROKER STREQUAL "")
            message(FATAL_ERROR "A temporary USB LAN candidate requires the LAN-only benchmark build")
        endif()
        target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_USB_LAN_CANDIDATE=1)
        if("$ENV{SESAME_PHONE_RX_TRIAL}" STREQUAL "1")
            target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_PHONE_RX_TRIAL=1)
            if("$ENV{SESAME_CORE_RX_WITNESS}" STREQUAL "1")
                set(probe "${EXPERIMENT_ROOT}/companion/core_rx")
                if("$ENV{SESAME_READER_QOS_TRIAL}" STREQUAL "1")
                    target_sources(sunshine_stream_exp PRIVATE "${probe}/reader_qos.cpp")
                    target_link_libraries(sunshine_stream_exp PRIVATE qos)
                    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_READER_QOS_TRIAL=1)
                endif()
                set(copies "${CMAKE_CURRENT_BINARY_DIR}/numeric-core-rx")
                find_program(SR_PYTHON NAMES python.exe python3 REQUIRED)
                set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${probe}/prepare.py")
                set(fec_trial_args "")
                if("$ENV{SESAME_KEY_RECOVERY_TRIAL}" STREQUAL "1")
                    list(APPEND fec_trial_args --guarded-retry)
                    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_KEY_RECOVERY_TRIAL=1)
                    if("$ENV{SESAME_KEY_SLOT_TRIAL}" STREQUAL "1")
                        target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_KEY_SLOT_TRIAL=1)
                    endif()
                endif()
                if("$ENV{SESAME_CORE_NORMAL_FEC}" STREQUAL "1")
                    list(APPEND fec_trial_args --normal-fec)
                endif()
                execute_process(COMMAND "${SR_PYTHON}" -B "${probe}/prepare.py" "${CORE_DIR}/src" "${copies}" ${fec_trial_args}
                    RESULT_VARIABLE probe_error)
                if(NOT probe_error EQUAL 0)
                    message(FATAL_ERROR "Pinned core observer generation failed")
                endif()
                get_target_property(core_original moonlight-common-c SOURCES)
                set(core_instrumented "")
                set(probe_count 0)
                foreach(unit IN LISTS core_original)
                    get_filename_component(leaf "${unit}" NAME)
                    if(leaf STREQUAL "VideoStream.c" OR leaf STREQUAL "VideoDepacketizer.c" OR leaf STREQUAL "RtpVideoQueue.c" OR leaf STREQUAL "ControlStream.c")
                        list(APPEND core_instrumented "${copies}/${leaf}")
                        math(EXPR probe_count "${probe_count}+1")
                    else()
                        list(APPEND core_instrumented "${unit}")
                    endif()
                endforeach()
                if(NOT probe_count EQUAL 4)
                    message(FATAL_ERROR "Unexpected pinned core target layout")
                endif()
                set_property(TARGET moonlight-common-c PROPERTY SOURCES "${core_instrumented}")
                target_include_directories(moonlight-common-c PRIVATE "${probe}")
                target_include_directories(sunshine_stream_exp PRIVATE "${probe}")
                target_sources(sunshine_stream_exp PRIVATE "${probe}/sesame_core_rx.cpp")
                target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_CORE_RX_WITNESS=1)
            endif()
        endif()
        target_include_directories(sunshine_stream_exp PRIVATE "${EXPERIMENT_ROOT}/companion")
    endif()
    if(EXISTS "${EXPERIMENT_ROOT}/.generated/native-route.tag")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${EXPERIMENT_ROOT}/.generated/native-route.tag")
    endif()
    if(NOT COMPANION_BROKER STREQUAL "")
        if(NOT COMPANION_BROKER MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+\\.[0-9]+:23388$")
            message(FATAL_ERROR "The companion test needs one explicit IPv4 UDP 23388 endpoint.")
        endif()
        target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_COMPANION_WAN=1
            "SESAME_SUNSHINE_COMPANION_BROKER=\"${COMPANION_BROKER}\"")
    endif()
endif()
target_compile_features(sunshine_stream_exp PRIVATE cxx_std_17)
target_link_options(sunshine_stream_exp PRIVATE -Wl,--exclude-libs,ALL
    -Wl,--wrap=pollSockets -Wl,--wrap=recv -Wl,--wrap=send)
if("$ENV{SESAME_SUNSHINE_LONG_BENCH}" STREQUAL "1" AND "$ENV{SESAME_SUNSHINE_COMPANION}" STREQUAL "1")
    target_compile_definitions(sunshine_stream_exp PRIVATE SESAME_SUNSHINE_NARROW_TRIAL=1)
    target_link_options(sunshine_stream_exp PRIVATE -Wl,--wrap=getSdpPayloadForStreamConfig)
endif()
target_link_libraries(sunshine_stream_exp PRIVATE sunshine_wan_peer moonlight-common-c ace_napi.z net_connection
    native_media_codecbase native_media_core native_media_vdec native_window hilog_ndk.z)
if("$ENV{SESAME_SUNSHINE_COMPANION}" STREQUAL "1")
    target_link_libraries(sunshine_stream_exp PRIVATE "${OPENSSL_SSL_LIBRARY}" "${OPENSSL_CRYPTO_LIBRARY}")
endif()
